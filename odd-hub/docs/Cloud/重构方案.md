### 6.2 Pipeline 处理流程

```delphi
function TRequestPipeline.Process(var AContext: TPipelineContext): TPipelineResult;
begin
  // 第一重：绝对门禁
  if psAbsoluteGate in FStages then
  begin
    GateResult := FAbsoluteGate.Check(AContext.Messages);
    if not GateResult.Passed then
    begin
      Result.Success := False;
      Result.ErrorMessage := GateResult.Reason;
      FAuditLog.Log(GateResult.AuditEntry);
      Exit;
    end;
  end;
  
  // 第二重：质疑引擎
  if psChallenge in FStages then
  begin
    ChallengeResult := FChallengeEngine.Analyze(AContext.Messages);
    if not ChallengeResult.IsValid then
    begin
      Result.Success := False;
      Result.ErrorMessage := ChallengeResult.Reason;
      Exit;
    end;
  end;
  
  // 原有阶段...
  // Auth �?Sanitize �?RiskGuard �?GoalInject �?Compress �?SelectAcct �?Forward
  
  // 第三重：响应门禁
  if psResponseGate in FStages then
  begin
    ValidationResponse := FResponseGate.Validate(AContext.Response);
    if not ValidationResponse.IsValid then
    begin
      Result.Success := False;
      Result.ErrorMessage := ValidationResponse.Reason;
      FAuditLog.Log(ValidationResponse.AuditEntry);
      Exit;
    end;
  end;
  
  Result.Success := True;
end;
```
