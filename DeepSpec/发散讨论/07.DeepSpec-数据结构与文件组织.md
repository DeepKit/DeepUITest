## 3. 通用 Spec 头部

每个 Spec YAML 都应有统一头部。

```yaml
meta:
  spec_id: "03.feature-tree"
  spec_type: "feature_tree"
  title: "功能树"
  status: "draft"
  version: "0.1"
  owner: "马三 付"
  created_at: "2026-05-13"
  updated_at: "2026-05-13"
  frozen_at: null
  current_hash: ""
  frozen_hash: ""

review:
  reviewer: null
  reviewed_at: null
  review_result: null
  review_notes: []

relations:
  upstream_specs: []
  downstream_specs: []
  related_files: []
  related_tests: []
  related_decisions: []

ai:
  ai_generated: false
  ai_last_run_id: null
  human_review_required: true
  ai_constraints: []
```

## 8. 状态机 Schema

```yaml

  states:
    - id: "empty"
      name: "空文档"
    - id: "draft"
      name: "草稿"
    - id: "reviewing"
      name: "待审阅"
    - id: "frozen"
      name: "已冻结"
    - id: "changing"
      name: "变更中"
    - id: "conflict"
      name: "冲突"
  transitions:
    - from: "draft"
      to: "reviewing"
      trigger: "submit_review"
    - from: "reviewing"
      to: "frozen"
      trigger: "human_confirm"
  forbidden_transitions:
    - from: "draft"
      to: "frozen"
      reason: "必须经过人类审阅"
```
