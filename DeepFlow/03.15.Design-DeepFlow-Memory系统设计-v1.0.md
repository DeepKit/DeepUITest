# DeepFlow Memory 系统设计

> 详细定义 DeepFlow 的三层记忆架构、向量检索方案、容量管理与淘汰策略。

创建日期: 2025-12-04  版本: 1.0

---

## 1. 概述

Memory 系统是 AI Agent 的核心能力之一，负责管理上下文、存储知识、支持回忆。DeepFlow 采用三层记忆架构，平衡性能与持久化需求。

### 1.1 设计目标

- **上下文连贯**：跨轮对话保持上下文
- **知识积累**：用户偏好和历史经验可学习
- **高效检索**：语义相关内容快速召回
- **容量可控**：防止无限增长

## 2. 三层记忆架构

### 2.1 Working Memory（工作记忆）

**定位**：当前任务执行期间的短期存储

**特点**：
- 生命周期：单次请求 Workflow 执行期间
- 存储位置：内存
- 容量：小于 1MB
- 访问速度：~1ms

**存储内容**：
```pascal
TWorkingMemory = class
private
  FCurrentTaskId: string;           // 当前任务 ID
  FVariables: TDictionary<string, TJSONValue>;  // 临时变量
FExecutionStack: TStack<TStepContext>;        // 执行栈
  FIntermediateResults: TList<TJSONObject>;     // 中间结果
  FStartTime: TDateTime;
public
  procedure Set(const AKey: string; AValue: TJSONValue);
  function Get(const AKey: string): TJSONValue;
  procedure Push(AContext: TStepContext);
  function Pop: TStepContext;
procedure Clear;  // 任务结束时调用
end;
```

**使用场景**：
```pascal
// Workflow 执行中传递变量
WorkingMemory.Set('user_intent', TJSONString.Create('shotlist'));
WorkingMemory.Set('extracted_entities', LEntitiesJSON);

// 后续步骤读取
LIntent := WorkingMemory.Get('user_intent').Value;
```

### 2.2 Session Memory（会话记忆）

**定位**：单次会话（对话）期间的上下文

**特点**：
- 生命周期：会话期间（分钟到小时）
- 存储位置：内存+ 可选持久化
- 容量：中等（< 100KB/会话）
- 访问速度：5ms

**存储内容**：

**上下文窗口管理**
```pascal
function TSessionMemory.GetContextWindow(AMaxTokens: Integer): TArray<TChatMessage>;
var
  LTotalTokens: Integer;
  I: Integer;
begin
  // 从最新消息开始，向前累加直到达到 token 限制
  LTotalTokens := 0;
  SetLength(Result, 0);
  
  for I := FMessages.Count - 1 downto 0 do
  begin
    if LTotalTokens + FMessages[I].TokenCount > AMaxTokens then
      Break;
    
    // 插入到结果开头
    SetLength(Result, Length(Result) + 1);
    Move(Result[0], Result[1], (Length(Result) - 1) * SizeOf(TChatMessage));
    Result[0] := FMessages[I];
    
    LTotalTokens := LTotalTokens + FMessages[I].TokenCount;
  end;
end;
```

### 3.2 向量存储选型

| 方案 | 适用场景 | 优点 | 缺点 | MVP 支持 |
|------|----------|------|------|----------|
| **SQLite + FTS5** | 小规模本地库 | 零依赖、简洁 | 非真向量搜索 | 无 |
| **Chroma** | 中规模本地库 | 开源、易用 | 需 Python | 无 |

### 3.3 MVP 向量存储实现

```pascal
// 使用 SQLite 存储 + 简单余弦相似度
TSQLiteVectorStore = class(TInterfacedObject, IVectorStore)
private
  FConnection: TSQLiteConnection;
  FEmbeddingProvider: IEmbeddingProvider;
  
  function CosineSimilarity(const A, B: TArray<Single>): Double;
public
  constructor Create(const ADbPath: string; AEmbedding: IEmbeddingProvider);
  
  procedure Add(const AId: string; const AContent: string; 
    const ATags: TArray<string>);
  
  function Search(const AQuery: string; ATopK: Integer): TArray<TSearchResult>;
  
  procedure Delete(const AId: string);
end;

// 数据库表结构
const
  CREATE_MEMORY_TABLE = '''
    CREATE TABLE IF NOT EXISTS memory_entries (
      id TEXT PRIMARY KEY,
      user_id TEXT NOT NULL,
      content TEXT NOT NULL,
      embedding BLOB NOT NULL,          -- 序列化的 float 数组
      tags TEXT,                        -- JSON 数组
      importance REAL DEFAULT 0.5,
      access_count INTEGER DEFAULT 0,
      last_accessed_at TEXT,
      created_at TEXT NOT NULL,
      metadata TEXT                     -- JSON
    );
    
    CREATE INDEX IF NOT EXISTS idx_memory_user ON memory_entries(user_id);
    CREATE INDEX IF NOT EXISTS idx_memory_importance ON memory_entries(importance DESC);
  ''';

function TSQLiteVectorStore.Search(const AQuery: string; 
  ATopK: Integer): TArray<TSearchResult>;
var
  LQueryEmbedding: TArray<Single>;
  LAllEntries: TArray<TMemoryEntry>;
  LScores: TArray<TPair<string, Double>>;
  I: Integer;
begin
  // 1. 生成查询向量
  LQueryEmbedding := FEmbeddingProvider.Embed(AQuery);
  
  // 2. 加载所有条目（MVP 简单实现，后续优化）
  LAllEntries := LoadAllEntries;
  
  // 3. 计算相似度
  SetLength(LScores, Length(LAllEntries));
  for I := 0 to High(LAllEntries) do
  begin
    LScores[I].Key := LAllEntries[I].Id;
    LScores[I].Value := CosineSimilarity(LQueryEmbedding, LAllEntries[I].Embedding);
  end;
  
  // 4. 排序至 TopK
  TArray.Sort<TPair<string, Double>>(LScores, 
    TComparer<TPair<string, Double>>.Construct(
      function(const A, B: TPair<string, Double>): Integer
      begin
        Result := Sign(B.Value - A.Value);  // 降序
      end
    )
  );
  
  SetLength(Result, Min(ATopK, Length(LScores)));
  for I := 0 to High(Result) do
  begin
    Result[I].Id := LScores[I].Key;
    Result[I].Score := LScores[I].Value;
    Result[I].Entry := GetEntryById(LScores[I].Key);
  end;
end;
```

### 3.4 混合检索策略

```pascal
// 结合关键词搜索和向量搜索
function TLongTermMemory.HybridSearch(const AQuery: string; 
  ATopK: Integer): TArray<TMemoryEntry>;
var
  LVectorResults: TArray<TSearchResult>;
  LKeywordResults: TArray<TSearchResult>;
  LMerged: TDictionary<string, Double>;
  LEntry: TSearchResult;
begin
  // 1. 向量搜索
  LVectorResults := FVectorStore.Search(AQuery, ATopK * 2);
  
// 2. 关键词搜索（FTS5引擎）
  LKeywordResults := FDatabase.FullTextSearch(AQuery, ATopK * 2);
  
// 3. 融合分数（RRF - Reciprocal Rank Fusion算法）
  LMerged := TDictionary<string, Double>.Create;
  try
    for LEntry in LVectorResults do
      LMerged.AddOrSetValue(LEntry.Id, 
        LMerged.GetValueOrDefault(LEntry.Id, 0) + 1 / (60 + LEntry.Rank));
    
    for LEntry in LKeywordResults do
      LMerged.AddOrSetValue(LEntry.Id, 
        LMerged.GetValueOrDefault(LEntry.Id, 0) + 1 / (60 + LEntry.Rank));
    
    // 4. 按融合分数排序返回TopK
    Result := GetTopKByScore(LMerged, ATopK);
  finally
    LMerged.Free;
  end;
end;
```

### 4.2 淘汰策略

```pascal
TEvictionPolicy = (
  epLRU,           // 最近最少使用
  epLFU,           // 最少使用频率
  epFIFO,          // 先进先出
  epImportance,    // 按重要性
  epHybrid         // 综合策略
);

// 综合淘汰策略
function TMemoryManager.CalculateEvictionScore(AEntry: TMemoryEntry): Double;
var
  LRecencyScore: Double;
  LFrequencyScore: Double;
  LImportanceScore: Double;
begin
  // 1. 时间衰减（越久分数越低）
  LRecencyScore := Exp(-DaysBetween(Now, AEntry.LastAccessedAt) / 30);
  
  // 2. 访问频率（归一化）
  LFrequencyScore := Min(AEntry.AccessCount / 100, 1.0);
  
  // 3. 重要性（用户标记或推断）
  LImportanceScore := AEntry.Importance;
  
  // 加权求和
  Result := 0.3 * LRecencyScore + 0.3 * LFrequencyScore + 0.4 * LImportanceScore;
end;

procedure TMemoryManager.Evict(ATargetCount: Integer);
var
  LEntries: TArray<TMemoryEntry>;
  LScores: TArray<TPair<string, Double>>;
  I: Integer;
begin
  LEntries := FLongTermMemory.GetAllEntries;
  SetLength(LScores, Length(LEntries));
  
  // 计算所有条目的淘汰分数
  for I := 0 to High(LEntries) do
  begin
    LScores[I].Key := LEntries[I].Id;
    LScores[I].Value := CalculateEvictionScore(LEntries[I]);
  end;
  
  // 按分数升序排序（低分先淘汰）
  TArray.Sort<TPair<string, Double>>(LScores, ...);
  
  // 删除分数最低的 ATargetCount 个条目
  for I := 0 to ATargetCount - 1 do
    FLongTermMemory.Delete(LScores[I].Key);
end;
```

### 4.3 Session 清理

```pascal
procedure TSessionManager.CleanupExpiredSessions;
var
  LSession: TSessionMemory;
begin
  for LSession in FSessions.Values do
  begin
// 超过超时时间未活动
    if MinutesBetween(Now, LSession.LastActiveAt) > FQuota.SessionTimeout then
    begin
      // 可选：将重要内容转存到长期记忆
      if LSession.HasImportantContent then
        TransferToLongTerm(LSession);
      
      FSessions.Remove(LSession.SessionId);
      LSession.Free;
    end;
  end;
end;

function TSessionMemory.HasImportantContent: Boolean;
begin
// 判断标准
// 1. 用户明确表达的偏好
// 2. 成功完成的任务
// 3. 用户正面反馈的内容
  Result := (FIntentHiDeepStory.Count > 0) and 
            (FIntentHiDeepStory.Any(function(I: TIntentRecord): Boolean
              begin Result := I.Resolved and (I.UserFeedback = ftPositive); end));
end;
```

```pascal
// Advisor 使用历史经验
function TAdvisor.GetRelevantExperiences(const ASituation: string): string;
var
  LExperiences: TArray<TExperience>;
  LPromptPart: string;
begin
  LExperiences := FMemory.GetSimilarExperiences(ASituation);
  
  if Length(LExperiences) = 0 then
    Exit('');
  
LPromptPart := '基于历史经验参考' + sLineBreak;
  for var Exp in LExperiences do
  begin
    if Exp.UserFeedback = ftPositive then
      LPromptPart := LPromptPart + Format('- 在类似情况%s"下，采取"%s"效果良好', 

        [Exp.Situation, Exp.Action]) + sLineBreak
    else if Exp.UserFeedback = ftNegative then
      LPromptPart := LPromptPart + Format('- 避免：在"%s"情况下采取%s"（效果不佳）', 

        [Exp.Situation, Exp.Action]) + sLineBreak;
  end;
  
  Result := LPromptPart;
end;
```
