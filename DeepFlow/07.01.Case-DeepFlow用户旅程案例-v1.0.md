#### Python Skill �?

```python
# gen_script.py - 脚本生成 Skill
from fastapi import FastAPI
from fastapi.responses import StreamingResponse
from litellm import completion
import json

class ScriptGeneratorSkill(BaseSkill):
    def get_meta(self) -> SkillMeta:
        return SkillMeta(
            name="gen_script",
            version="1.0.0",
            description="Generate video script using LLM",
            input_schema={"topic": "string", "duration": "integer"},
            output_schema={"script": "string", "scenes": "array"}
        )
    
    async def stream(self, input: SkillInput):
        """流式生成脚本"""
        topic = input.data.get("topic")
        duration = input.data.get("duration", 60)
        
        prompt = f"""为以下主题生成一个{duration}秒的视频脚本�?
主题：{topic}

要求�?
1. 包含开场、主体、结�?
2. 每个场景标注时长
3. 包含画面描述和旁白文�?
"""
        
        # 调用 LiteLLM 流式接口
        response = completion(
            model="openrouter/anthropic/claude-3-sonnet",
            messages=[{"role": "user", "content": prompt}],
            stream=True
        )
        
        full_content = ""
        for chunk in response:
            if chunk.choices[0].delta.content:
                content = chunk.choices[0].delta.content
                full_content += content
                yield json.dumps({
                    "type": "chunk",
                    "content": content
                })
        
        # 发送完成信�?
        yield json.dumps({
            "type": "done",
            "full_content": full_content
        })

# FastAPI 端点
```

#### Delphi SignalOfficer �?

```pascal
// SignalOfficer 处理 SSE 流式响应
procedure TSignalOfficer.StreamRequest(const AUrl: string;
  const ABody: TJSONObject; const ACallback: TStreamCallback);
var
  LClient: THTTPClient;
  LResponse: IHTTPResponse;
  LReader: TStreamReader;
  LLine: string;
begin
  LClient := THTTPClient.Create;
  try
    // 设置 SSE �?
    LClient.ContentType := 'application/json';
    LClient.Accept := 'text/event-stream';
    
    // 发起请求
    LResponse := LClient.Post(AUrl, ABody.ToJSON);
    
    // 逐行读取 SSE 数据
    LReader := TStreamReader.Create(LResponse.ContentStream);
    try
      while not LReader.EndOfStream do
      begin
        LLine := LReader.ReadLine;
        if LLine.StartsWith('data: ') then
        begin
          var LData := LLine.Substring(6);
          var LJson := TJSONObject.ParseJSONValue(LData) as TJSONObject;
          
          if LJson.GetValue<string>('type') = 'chunk' then
            ACallback(LJson.GetValue<string>('content'), False)
          else if LJson.GetValue<string>('type') = 'done' then
          begin
            ACallback('', True);
            Break;
          end;
        end;
      end;
    finally
      LReader.Free;
    end;
  finally
    LClient.Free;
  end;
end;
```
