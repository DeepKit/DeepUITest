```ts
  request: CreateChildFlowRequest

    throw new Error(`MAX_CHILDREN_EXCEEDED: max=${NESTING_CONSTRAINTS.maxChildrenPerFlow}`);

    flowId: generateFlowId(),
    type: request.type,
    status: 'Created',
    parentFlowId: parent.flowId,
    rootFlowId: parent.rootFlowId ?? parent.flowId,
    depth: parent.depth + 1,
    childFlowIds: [],
  };
  
  // 3. 更新父流�?
  await updateFlow(parent.flowId, {
    childFlowIds: [...parent.childFlowIds, childFlow.flowId],
  });
  
  // 4. 记录创建事件
  await writeEvent({
    flowId: childFlow.flowId,
    type: request.type,
    step: '__lifecycle__',
    source: 'uniFlow/FlowManager',
    status: 'Started',
    payload: {
      action: 'CHILD_FLOW_CREATED',
      parentFlowId: parent.flowId,
      rootFlowId: childFlow.rootFlowId,
      depth: childFlow.depth,
    },
  });
  
  return childFlow;
}
```

```ts
async function executeCompensation(

  const config = getFlowConfig(parentFlow.type);

  return result.status === 'Succeeded';
```

```ts
// Saga 模式实现
interface SagaStep {
  name: string;
  flowType: UniFlowType;
  compensationFlowType?: UniFlowType;
}

class SagaOrchestrator {
  private steps: SagaStep[];
  private completedSteps: Map<string, UniFlowId> = new Map();
  
  async execute(parentFlowId: UniFlowId): Promise<void> {
    for (const step of this.steps) {
      try {
        const childFlow = await this.executeStep(parentFlowId, step);
        this.completedSteps.set(step.name, childFlow.flowId);
      } catch (error) {
        await this.compensate();
        throw error;
      }
    }
  }
  
  private async compensate(): Promise<void> {
    // 逆序执行补偿
    const completedStepNames = Array.from(this.completedSteps.keys()).reverse();
    
    for (const stepName of completedStepNames) {
      const step = this.steps.find(s => s.name === stepName);
      if (step?.compensationFlowType) {
        await this.executeCompensationStep(step);
      }
    }
  }
}
```

## 5. 并行子流�?

### 5.1 并行执行模型

```ts
interface ParallelExecutionConfig {
  maxConcurrency: number;        // 最大并发数
  failureThreshold?: number;     // 允许失败数（超过则整体失败）
  timeout?: number;              // 整体超时时间
}

async function executeParallelChildFlows(
  parentFlowId: UniFlowId,
  childConfigs: CreateChildFlowRequest[],
  config: ParallelExecutionConfig
): Promise<ParallelExecutionResult> {
  const semaphore = new Semaphore(config.maxConcurrency);
  const results: ChildFlowResult[] = [];
  let failureCount = 0;
  
  const promises = childConfigs.map(async (childConfig) => {
    await semaphore.acquire();
    
    try {
      const childFlow = await createChildFlow({
        ...childConfig,
        parentFlowId,
      });
      
      const result = await waitForFlow(childFlow.flowId, config.timeout);
      results.push({ flowId: childFlow.flowId, result });
      
      if (result.status === 'Failed') {
        failureCount++;
        if (config.failureThreshold && failureCount > config.failureThreshold) {
          throw new Error('PARALLEL_FAILURE_THRESHOLD_EXCEEDED');
        }
      }
    } finally {
      semaphore.release();
    }
  });
  
  await Promise.all(promises);
  
  return {
    total: childConfigs.length,
    succeeded: results.filter(r => r.result.status === 'Succeeded').length,
    failed: failureCount,
    results,
  };
}
```

```ts
interface NestedEventQuery {
  // 查询整棵树的所有事�?
  rootFlowId: UniFlowId;
  includeChildren?: boolean;
  maxDepth?: number;
  
  // 过滤条件
  flowTypes?: UniFlowType[];
  status?: ('Started' | 'Succeeded' | 'Failed')[];
  timeRange?: { from: Date; to: Date };
}

async function queryNestedEvents(
  query: NestedEventQuery
): Promise<NestedFlowEvent[]> {
  // 递归查询所有子流程事件
  const events: NestedFlowEvent[] = [];
  
  async function collectEvents(flowId: UniFlowId, depth: number) {
    if (query.maxDepth !== undefined && depth > query.maxDepth) return;
    
    const flow = await getFlow(flowId);
    const flowEvents = await getEvents(flowId, query);
    
    events.push(...flowEvents.map(e => ({
      ...e,
      nestingContext: {
        rootFlowId: query.rootFlowId,
        parentFlowId: flow.parentFlowId,
        depth,
        path: await buildFlowPath(flowId),
      },
    })));
    
    if (query.includeChildren) {
      for (const childId of flow.childFlowIds) {
        await collectEvents(childId, depth + 1);
      }
    }
  }
  
  await collectEvents(query.rootFlowId, 0);
  return events;
}
```

## 7. 嵌套快照管理

### 7.1 组合快照

```ts
interface CompositeSnapshot {
  rootFlowId: UniFlowId;
  rootSnapshot: UniFlowSnapshot;
  childSnapshots: Map<UniFlowId, UniFlowSnapshot>;
  
  // 聚合视图
  aggregateState: any;
}

async function buildCompositeSnapshot(
  rootFlowId: UniFlowId
): Promise<CompositeSnapshot> {
  const rootFlow = await getFlow(rootFlowId);
  const rootSnapshot = await getSnapshot(rootFlowId);
  const childSnapshots = new Map<UniFlowId, UniFlowSnapshot>();
  
  // 递归收集子流程快�?
  async function collectSnapshots(flowId: UniFlowId) {
    const flow = await getFlow(flowId);
    for (const childId of flow.childFlowIds) {
      const childSnapshot = await getSnapshot(childId);
      childSnapshots.set(childId, childSnapshot);
      await collectSnapshots(childId);
    }
  }
  
  await collectSnapshots(rootFlowId);
  
  return {
    rootFlowId,
    rootSnapshot,
    childSnapshots,
    aggregateState: mergeSnapshots(rootSnapshot, childSnapshots),
  };
}
```
