```pascal
    Data: T;

begin
  Result.Success := True;
  Result.Data := AData;
  Result.Error := Default(TErrorInfo);
end;

class function TResult<T>.Fail(AError: TErrorInfo): TResult<T>;
begin
  Result.Success := False;
  Result.Data := Default(T);
  Result.Error := AError;
end;

  Names: array[TRoleType] of string = (
    'engine', 'inspector', 'commander', 'dispatcher', 'executor',
    'advisor', 'guard', 'quartermaster', 'logistics', 'chronicler',
```

```pascal
constructor TCommander.Create(AEngine: IEngine; AConfig: TJSONObject);
begin
  inherited;
  FRoleType := rtCommander;
end;

procedure TCommander.Initialize;
begin
  inherited;
  // 获取依赖角色引用
  FGuard := FEngine.GetRole<IGuard>(rtGuard);
  FDispatcher := FEngine.GetRole<IDispatcher>(rtDispatcher);
  Log(llInfo, 'Commander initialized');
end;

procedure TCommander.HandleMessage(AMessage: TUniFlowMessage);
begin
  case AMessage.Body.Action of
    'HANDLE_REQUEST':
      begin
        var LResult := HandleRequest(AMessage.Body.Payload);
        SendMessage(CreateResponse(AMessage, LResult.ToJSON));
      end;
  else
    Log(llWarn, 'Unknown action: ' + AMessage.Body.Action);
  end;
end;

var
  LValidation: TResult<Boolean>;
```

```pascal
  FResponses.Free;

  FResponses.Enqueue(AResponse);

  Result := False;

    if Msg.Body.Action = AAction then
      Exit(True);

    Result := TResult<Boolean>.Ok(True);

  if FCurrentFails < FFailCount then
  begin
    Inc(FCurrentFails);

      'Mock failure', True)));
  end;
  
  if Assigned(FNextResponse) then
    Result := TResult<TJSONObject>.Ok(FNextResponse)
  else
    Result := TResult<TJSONObject>.Ok(TMockFactory.CreateLLMResponse('Mock response'));
end;

end.
```

## 6. Python Skills 结构

### 6.1 Skill 基类

```python
# skills/uniflow/skill.py
from abc import ABC, abstractmethod
from typing import Dict, Any, Optional
from pydantic import BaseModel

class SkillResult(BaseModel):
    success: bool
    data: Optional[Dict[str, Any]] = None
    error: Optional[Dict[str, Any]] = None

def skill_result(success: bool, data: dict = None, error: dict = None) -> dict:
    return SkillResult(success=success, data=data, error=error).dict()

class Skill(ABC):
    """Skill 基类"""
    
    skill_id: str = "base"
    description: str = "Base skill"
    
    def __init__(self, advisor_client, config: dict = None):
        self.advisor = advisor_client
        self.config = config or {}
    
    @abstractmethod
    async def execute(self, input: dict, context: dict) -> dict:
        """执行 Skill 逻辑"""
        pass
    
    def validate_input(self, input: dict) -> bool:
        """验证输入"""
        return True
    
    @property
    def metadata(self) -> dict:
        return {
            "id": self.skill_id,
            "description": self.description
        }
```

### 6.2 FastAPI 入口

```python
# skills/main.py
from fastapi import FastAPI, HTTPException
from pydantic import BaseModel
from typing import Dict, Any, Optional
import importlib
import os

app = FastAPI(title="UniFlow Skills Service")

# Skill 注册�?
skill_registry: Dict[str, "Skill"] = {}

class SkillRequest(BaseModel):
    skill_id: str
    input: Dict[str, Any]
    context: Optional[Dict[str, Any]] = {}

class SkillResponse(BaseModel):
    success: bool
    data: Optional[Dict[str, Any]] = None
    error: Optional[Dict[str, Any]] = None

@app.on_event("startup")
async def startup():
    """启动时注册所�?Skills"""
    from skills import decision_coach, critic, echo
    
    for skill_class in [
        decision_coach.DecisionCoachSkill,
        critic.CriticSkill,
        echo.EchoSkill,
    ]:
        instance = skill_class(advisor_client=None)
        skill_registry[instance.skill_id] = instance
    
    print(f"Registered {len(skill_registry)} skills")

@app.get("/health")
async def health():
    return {"status": "healthy", "skills": len(skill_registry)}

@app.get("/skills")
async def list_skills():
    return [s.metadata for s in skill_registry.values()]

@app.post("/execute", response_model=SkillResponse)
async def execute_skill(request: SkillRequest):
    skill = skill_registry.get(request.skill_id)
    if not skill:
        raise HTTPException(404, f"Skill not found: {request.skill_id}")
    
    try:
        result = await skill.execute(request.input, request.context)
        return SkillResponse(**result)
    except Exception as e:
        return SkillResponse(
            success=False,
            error={"code": "SKILL_ERROR", "message": str(e)}
        )

if __name__ == "__main__":
    import uvicorn
    uvicorn.run(app, host="0.0.0.0", port=8765)
```
