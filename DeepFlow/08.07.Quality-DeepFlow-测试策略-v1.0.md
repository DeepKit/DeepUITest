### 4.2 混沌测试框架

```typescript
interface ChaosEngine {
  // 故障注入
  injectFault(fault: FaultDefinition): FaultHandle
  
  // 移除故障
  removeFault(handle: FaultHandle): void
  
  // 清除所有故�?

}

interface FaultDefinition {
  type: 'delay' | 'error' | 'drop' | 'partition' | 'resource'
  target: {
    role?: RoleType
    action?: string
    message?: string

  config: {

}
```
