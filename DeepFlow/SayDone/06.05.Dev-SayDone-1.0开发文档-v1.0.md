unit NL.Cache;

interface

uses
  System.SysUtils, System.Generics.Collections, System.Hash,
  Domain.NL, Infra.Sqlite.NLCacheRepository;

type
  TCacheLevel = (clExact, clSemantic, clPattern, clReference);

  TCacheResult = record
    Level: TCacheLevel;
    Data: TNLParseResult;
    Confidence: Double;
    References: TArray<TNLParseResult>;  // Level D 参数
  end;

  INLCache = interface
    ['{D4E5F6A7-B8C9-0123-DEFG-456789012345}']
    function Query(const AInput, ASceneType: string): TCacheResult;
    procedure Store(const AInput, ASceneType: string; const AResult: TNLParseResult);
  end;

  TNLCache = class(TInterfacedObject, INLCache)
  private
    FRepo: INLCacheRepository;
    
    function QueryLevelA(const AInput, ASceneType: string): TCacheResult;
    function QueryLevelB(const AInput, ASceneType: string): TCacheResult;
    function QueryLevelC(const AInput, ASceneType: string): TCacheResult;
    function QueryLevelD(const AInput, ASceneType: string): TCacheResult;
    
    function ComputeHash(const AInput, ASceneType: string): string;
    function ComputeSimilarity(const AInput1, AInput2: string): Double;
    function ExtractPattern(const AInput: string): string;
  public
    constructor Create(ARepo: INLCacheRepository);
    
    function Query(const AInput, ASceneType: string): TCacheResult;
    procedure Store(const AInput, ASceneType: string; const AResult: TNLParseResult);
  end;

implementation

function TNLCache.Query(const AInput, ASceneType: string): TCacheResult;
begin
  // Level A: 精确匹配（hash相同）
  Result := QueryLevelA(AInput, ASceneType);
  if Result.Level = clExact then Exit;
  
  // Level B: 语义匹配（相似度 > 0.95）
  Result := QueryLevelB(AInput, ASceneType);
  if Result.Level = clSemantic then Exit;
  
  // Level C: 模式匹配（提取模板后匹配）
  Result := QueryLevelC(AInput, ASceneType);
  if Result.Level = clPattern then Exit;
  
  // Level D: 参考匹配（相似案例参考）
  Result := QueryLevelD(AInput, ASceneType);
end;

function TNLCache.QueryLevelA(const AInput, ASceneType: string): TCacheResult;
var
  LHash: string;
  LCached: TNLCacheRecord;
begin
  Result.Level := clExact;
  Result.Confidence := 0;
  
  LHash := ComputeHash(AInput, ASceneType);
  LCached := FRepo.GetByHash(LHash);
  
  if not LCached.IsEmpty then
  begin
    Result.Data := TNLParseResult.FromJSON(LCached.OutputJson);
    Result.Confidence := 1.0;
    FRepo.IncrementHitCount(LCached.Id);
  end;
end;

function TNLCache.QueryLevelB(const AInput, ASceneType: string): TCacheResult;
var
  LCandidates: TArray<TNLCacheRecord>;
  LCandidate: TNLCacheRecord;
  LSimilarity: Double;
begin
  Result.Level := clSemantic;
  Result.Confidence := 0;
  
  // 获取同场景的候选
  LCandidates := FRepo.GetBySceneType(ASceneType, 100);
  
  for LCandidate in LCandidates do
  begin
    LSimilarity := ComputeSimilarity(AInput, LCandidate.InputText);
    if LSimilarity > 0.95 then
    begin
      Result.Data := TNLParseResult.FromJSON(LCandidate.OutputJson);
      Result.Confidence := LSimilarity;
      FRepo.IncrementHitCount(LCandidate.Id);
      Exit;
    end;
  end;
end;

unit NL.ModelRouter;

interface

uses
  System.SysUtils, DeepBase.Config;

type
  TComplexityScore = record
    InputLength: Integer;
    EntityCount: Integer;
    RelationCount: Integer;
    Ambiguity: Double;
    Total: Double;
    
    class function Compute(const AInput: string): TComplexityScore; static;
  end;

  TModelConfig = record
    ModelName: string;
    MaxTokens: Integer;
    Temperature: Double;
  end;

  INLModelRouter = interface
    ['{E5F6A7B8-C9D0-1234-EFGH-567890123456}']
    function SelectModel(const AComplexity: TComplexityScore): TModelConfig;
  end;

  TNLModelRouter = class(TInterfacedObject, INLModelRouter)
  private
    FSimpleThreshold: Double;
    FMediumThreshold: Double;
  public
    constructor Create;
    function SelectModel(const AComplexity: TComplexityScore): TModelConfig;
  end;

implementation

class function TComplexityScore.Compute(const AInput: string): TComplexityScore;
begin
  Result.InputLength := Length(AInput);
  Result.EntityCount := 0;  // TODO: 实体识别
  Result.RelationCount := 0;
  Result.Ambiguity := 0.5;
  
  // 综合评分
  Result.Total := Result.InputLength * 0.1 + 
                  Result.EntityCount * 10 + 
                  Result.Ambiguity * 50;
end;

constructor TNLModelRouter.Create;
begin
  inherited;
  FSimpleThreshold := DeepBase.Config.GetConfigFloat('nl.router.simple_threshold', 20);
  FMediumThreshold := DeepBase.Config.GetConfigFloat('nl.router.medium_threshold', 60);
end;

function TNLModelRouter.SelectModel(const AComplexity: TComplexityScore): TModelConfig;
begin
  // 简单场景：本地模型
  if AComplexity.Total < FSimpleThreshold then
  begin
    Result.ModelName := 'ollama/qwen2.5:7b';
    Result.MaxTokens := 500;
    Result.Temperature := 0.1;
    Exit;
  end;
  
  // 中等场景：Claude Haiku / GPT-3.5
  if AComplexity.Total < FMediumThreshold then
  begin
    Result.ModelName := 'claude-3-haiku';
    Result.MaxTokens := 1000;
    Result.Temperature := 0.2;
    Exit;
  end;
  
  // 复杂场景：GPT-4
  Result.ModelName := 'gpt-4-turbo';
  Result.MaxTokens := 2000;
  Result.Temperature := 0.3;
end;

end.

unit NL.ValidationChain;

interface

uses
  System.SysUtils, System.Generics.Collections, Domain.NL;

type
  TValidationResult = record
    Valid: Boolean;
    Errors: TArray<string>;
    Warnings: TArray<string>;
    AutoFixable: Boolean;
    SuggestedFixes: TArray<string>;
  end;

  TValidationReport = record
    Valid: Boolean;
    Errors: TArray<string>;
    Warnings: TArray<string>;
    Fixes: TArray<string>;
  end;

  IValidator = interface
    function GetName: string;
    function Validate(const AData: TNLParseResult): TValidationResult;
    property Name: string read GetName;
  end;

  TValidationChain = class
  private
    FValidators: TList<IValidator>;
  public
    constructor Create;
    destructor Destroy; override;
    
    procedure AddValidator(AValidator: IValidator);
    function Validate(const AData: TNLParseResult): TValidationReport;
  end;

// 具体验证器。
  TSchemaValidator = class(TInterfacedObject, IValidator)
  public
    function GetName: string;
    function Validate(const AData: TNLParseResult): TValidationResult;
  end;

  TTypeValidator = class(TInterfacedObject, IValidator)
  public
    function GetName: string;
    function Validate(const AData: TNLParseResult): TValidationResult;
  end;

  TLogicValidator = class(TInterfacedObject, IValidator)
  public
    function GetName: string;
    function Validate(const AData: TNLParseResult): TValidationResult;
  end;

  TSemanticValidator = class(TInterfacedObject, IValidator)
  private
    FAiService: INLAiService;  // AI辅助语义验证
  public
    constructor Create(AAiService: INLAiService);
    function GetName: string;
    function Validate(const AData: TNLParseResult): TValidationResult;
  end;

implementation

constructor TValidationChain.Create;
begin
  inherited;
  FValidators := TList<IValidator>.Create;
  
  // 按顺序添加验证器
  FValidators.Add(TSchemaValidator.Create);
  FValidators.Add(TTypeValidator.Create);
  FValidators.Add(TLogicValidator.Create);
  // SemanticValidator 需要外部注入AiService
end;

function TValidationChain.Validate(const AData: TNLParseResult): TValidationReport;
var
  LValidator: IValidator;
  LResult: TValidationResult;
begin
  Result.Valid := True;
  Result.Errors := [];
  Result.Warnings := [];
  Result.Fixes := [];
  
  for LValidator in FValidators do
  begin
    LResult := LValidator.Validate(AData);
    
    if Length(LResult.Errors) > 0 then
    begin
      Result.Valid := False;
      Result.Errors := Result.Errors + LResult.Errors;
      
      // 尝试自动修复
      if LResult.AutoFixable then
        Result.Fixes := Result.Fixes + LResult.SuggestedFixes;
    end;
    
    Result.Warnings := Result.Warnings + LResult.Warnings;
  end;
end;

// Schema验证器实现。
function TSchemaValidator.GetName: string;
begin
  Result := 'Schema';
end;

function TSchemaValidator.Validate(const AData: TNLParseResult): TValidationResult;
begin
  Result.Errors := [];
  Result.Warnings := [];
  Result.AutoFixable := False;
  
  // 必须有标题。
  if AData.Title.Trim.IsEmpty then
    Result.Errors := Result.Errors + ['缺少页面标题'];
    
  // 必须有字段。
  if Length(AData.Fields) = 0 then
    Result.Errors := Result.Errors + ['表单没有任何字段'];
    
  Result.Valid := Length(Result.Errors) = 0;
end;

end.

unit NL.TrustBank;

interface

uses
  System.SysUtils, System.Generics.Collections,

type
  INLTrustBank = interface
    ['{F6A7B8C9-D0E1-2345-FGHI-678901234567}']
    procedure RecordOutcome(const AUserId: string; ACorrect: Boolean);
    function GetTrustScore(const AUserId: string): Double;
    function ShouldAutoApply(const AUserId: string): Boolean;
    function ShouldAskFirst(const AUserId: string): Boolean;
  end;

  TNLTrustBank = class(TInterfacedObject, INLTrustBank)
  private
    FRepo: IPreferenceRepository;
    FAutoApplyThreshold: Double;
    FAskFirstThreshold: Double;
  public
    constructor Create(ARepo: IPreferenceRepository);
    
    procedure RecordOutcome(const AUserId: string; ACorrect: Boolean);
    function GetTrustScore(const AUserId: string): Double;
    function ShouldAutoApply(const AUserId: string): Boolean;
    function ShouldAskFirst(const AUserId: string): Boolean;
  end;

implementation

uses DeepBase.Config;

constructor TNLTrustBank.Create(ARepo: IPreferenceRepository);
begin
  inherited Create;
  FRepo := ARepo;
  FAutoApplyThreshold := DeepBase.Config.GetConfigFloat('nl.trust.auto_apply', 70);
  FAskFirstThreshold := DeepBase.Config.GetConfigFloat('nl.trust.ask_first', 30);
end;

procedure TNLTrustBank.RecordOutcome(const AUserId: string; ACorrect: Boolean);
var
  LCurrent, LDelta, LNew: Double;
begin
  LCurrent := FRepo.GetTrustScore(AUserId);
  if LCurrent < 0 then LCurrent := 50;  // 默认分数
  
  // 正确 +1，错误 -5（不对称惩罚机制）
  if ACorrect then
    LDelta := 1
  else
    LDelta := -5;
    
  LNew := Max(0, Min(100, LCurrent + LDelta));
  FRepo.SetTrustScore(AUserId, LNew);
end;

function TNLTrustBank.GetTrustScore(const AUserId: string): Double;
begin
  Result := FRepo.GetTrustScore(AUserId);
  if Result < 0 then Result := 50;
end;

function TNLTrustBank.ShouldAutoApply(const AUserId: string): Boolean;
begin
  Result := GetTrustScore(AUserId) >= FAutoApplyThreshold;
end;

function TNLTrustBank.ShouldAskFirst(const AUserId: string): Boolean;
begin
  Result := GetTrustScore(AUserId) < FAskFirstThreshold;
end;

end.

    FStageLabels[LStage].Text := STAGE_LABELS[LStage];
    FStageLabels[LStage].Opacity := 0.3;
    
    LTop := LTop + 30;
  end;
  
  FPartialContent := TLayout.Create(Self);
  FPartialContent.Parent := Self;
  FPartialContent.Position.Y := LTop + 20;
  FPartialContent.Align := TAlignLayout.Bottom;
end;

  FCurrentStage := AStage;
  UpdateStageUI;
  
  if not ADetail.IsEmpty then
    FStageLabels[AStage].Text := STAGE_LABELS[AStage] + ' ' + ADetail;

  LStage: TProgressStage;

  for LStage := Low(TProgressStage) to High(TProgressStage) do
  begin
    if LStage < FCurrentStage then
      FStageLabels[LStage].Opacity := 0.5
    else if LStage = FCurrentStage then
      FStageLabels[LStage].Opacity := 1.0
