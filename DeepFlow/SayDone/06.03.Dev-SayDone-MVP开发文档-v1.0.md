# SayDone MVP 开发文档

> 最小可行产品 - 验证核心概念
> 基于 DevDirector 架构（Delphi FMX + SQLite + liteLLM等）

---

## 1. MVP 目标

### 1.1 核心验证点

```
验证一个命题：
"用户说一句话，系统就能生成正确的结构化数据

MVP 只需要证明这是可行的
不需要做到完整
```

### 1.2 成功标准

| 指标 | 目标 |
|------|------|
| 首次正确率 | 180% |
| 交互次数 | 13次得到想要的结果 |
| 响应时间 | < 3秒|

### 4.1 NL解析服务

```delphi
unit NL.Parser;

interface

uses
  System.SysUtils, System.Classes, System.JSON, System.Threading,
  Domain.NL, NL.AiService, NL.PromptBuilder;

type
  INLParser = interface
    ['{A1B2C3D4-E5F6-7890-ABCD-EF1234567890}']
    procedure ParseAsync(
      const AInput: string;
      const ASceneType: string;
      const OnSuccess: TProc<TNLParseResult>;
      const OnError: TProc<string>
    );
  end;

  TNLParser = class(TInterfacedObject, INLParser)
  private
    FAiService: INLAiService;
    FPromptBuilder: TNLPromptBuilder;
  public
    constructor Create(AAiService: INLAiService);
    destructor Destroy; override;

    procedure ParseAsync(
      const AInput: string;
      const ASceneType: string;
      const OnSuccess: TProc<TNLParseResult>;
      const OnError: TProc<string>
    );
  end;

implementation

constructor TNLParser.Create(AAiService: INLAiService);
begin
  inherited Create;
  FAiService := AAiService;
  FPromptBuilder := TNLPromptBuilder.Create;
end;

destructor TNLParser.Destroy;
begin
  FPromptBuilder.Free;
  inherited;
end;

procedure TNLParser.ParseAsync(
  const AInput: string;
  const ASceneType: string;
  const OnSuccess: TProc<TNLParseResult>;
  const OnError: TProc<string>);
begin
  // 参数验证
  if AInput.Trim.IsEmpty then
  begin
    if Assigned(OnError) then OnError('输入不能为空');
    Exit;
  end;

  // 异步执行，避免阻塞UI
  TTask.Run(
    procedure
    var
      LPrompt: string;
      LResponse: string;
      LResult: TNLParseResult;
    begin
      try
        // 1. 构建 Prompt
        LPrompt := FPromptBuilder.Build(AInput, ASceneType);

        // 2. 调用 AI
        LResponse := FAiService.Chat(LPrompt);

        // 3. 解析 JSON 结果
        LResult := TNLParseResult.FromJSON(LResponse);

        // 4. 回调主线程
        TThread.Queue(nil,
          procedure
          begin
            if Assigned(OnSuccess) then OnSuccess(LResult);
          end);
      except
        on E: Exception do
          TThread.Queue(nil,
            procedure
            begin
              if Assigned(OnError) then OnError(E.Message);
            end);
      end;
    end);
end;

end.
```

### 4.2 Schema 验证服务

```delphi
unit NL.Validator;

interface

uses
  System.SysUtils, System.Generics.Collections, Domain.NL;

type
  TNLValidationResult = record
    Valid: Boolean;
    Errors: TArray<string>;
    Warnings: TArray<string>;
  end;

  INLValidator = interface
    ['{B2C3D4E5-F6A7-8901-BCDE-F23456789012}']
    function Validate(const AResult: TNLParseResult): TNLValidationResult;
  end;

  TNLValidator = class(TInterfacedObject, INLValidator)
  public
    function Validate(const AResult: TNLParseResult): TNLValidationResult;
  private
    function ValidateSchema(const AResult: TNLParseResult): TArray<string>;
    function ValidateLogic(const AResult: TNLParseResult): TArray<string>;
  end;

implementation

function TNLValidator.Validate(const AResult: TNLParseResult): TNLValidationResult;
var
  LSchemaErrors, LLogicErrors: TArray<string>;
begin
  // 1. Schema 验证（结构正确性）
  LSchemaErrors := ValidateSchema(AResult);

  // 2. 逻辑验证（语义合理性）
  LLogicErrors := ValidateLogic(AResult);

  Result.Errors := LSchemaErrors + LLogicErrors;
  Result.Valid := Length(Result.Errors) = 0;
end;

function TNLValidator.ValidateSchema(const AResult: TNLParseResult): TArray<string>;
const
  VALID_FIELD_TYPES: array[0..4] of string = 
    ('text', 'password', 'email', 'phone', 'textarea');
var
  LField: TNLField;
  LValidType: Boolean;
  I: Integer;
begin
  Result := [];

  // 必须有标题
  if AResult.Title.Trim.IsEmpty then
    Result := Result + ['缺少页面标题'];

  // 标题长度限制
  if Length(AResult.Title) > 50 then
    Result := Result + ['标题超过50字符'];

  // 字段类型验证
  for LField in AResult.Fields do
  begin
    LValidType := False;
    for I := Low(VALID_FIELD_TYPES) to High(VALID_FIELD_TYPES) do
      if LField.FieldType = VALID_FIELD_TYPES[I] then
      begin
        LValidType := True;
        Break;
      end;
    if not LValidType then
      Result := Result + [Format('无效字段类型: %s', [LField.FieldType])];
  end;
end;

function TNLValidator.ValidateLogic(const AResult: TNLParseResult): TArray<string>;
var
  LField: TNLField;
  LHasPassword: Boolean;
begin
  Result := [];

  // 登录页必须有密码字段
  if AResult.Title.Contains('登录') then
  begin
    LHasPassword := False;
    for LField in AResult.Fields do
      if LField.FieldType = 'password' then
      begin
        LHasPassword := True;
        Break;
      end;
    if not LHasPassword then
      Result := Result + ['登录页缺少密码字段'];
  end;

  // 表单必须有字段
  if Length(AResult.Fields) = 0 then
    Result := Result + ['表单没有任何字段'];
end;

end.
```

### 4.3 Prompt 构建器（从数据库加载）

> **少编译原则**：提示词存储在数据库，不硬编码到源代码中

```delphi
unit NL.PromptBuilder;

interface

uses
  System.SysUtils, Infra.Sqlite.PromptRepository;

type
  INLPromptBuilder = interface
    ['{C3D4E5F6-A7B8-9012-CDEF-345678901234}']
    function Build(const AInput, ASceneType: string): string;
    function GetSystemPrompt(const ASceneType: string): string;
  end;

  TNLPromptBuilder = class(TInterfacedObject, INLPromptBuilder)
  private
    FPromptRepo: IPromptRepository;
    FCache: TDictionary<string, string>;  // 内存缓存
  public
    constructor Create(APromptRepo: IPromptRepository);
    destructor Destroy; override;
    
    function Build(const AInput, ASceneType: string): string;
    function GetSystemPrompt(const ASceneType: string): string;
  end;

implementation

constructor TNLPromptBuilder.Create(APromptRepo: IPromptRepository);
begin
  inherited Create;
  FPromptRepo := APromptRepo;
  FCache := TDictionary<string, string>.Create;
end;

destructor TNLPromptBuilder.Destroy;
begin
  FCache.Free;
  inherited;
end;

function TNLPromptBuilder.GetSystemPrompt(const ASceneType: string): string;
var
  LPromptRec: TPromptRecord;
begin
  // 1. 先查内存缓存
  if FCache.TryGetValue(ASceneType, Result) then
    Exit;
    
  // 2. 从数据库加载（带版本控制）
  LPromptRec := FPromptRepo.GetActivePrompt('nl_system', ASceneType);
  if LPromptRec.Content.IsEmpty then
    raise Exception.CreateFmt('未找到场景[%s] 的提示词配置', [ASceneType]);
    
  Result := LPromptRec.Content;
  
// 3. 缓存到内存
  FCache.AddOrSetValue(ASceneType, Result);
end;

function TNLPromptBuilder.Build(const AInput, ASceneType: string): string;
begin
  Result := GetSystemPrompt(ASceneType) + sLineBreak + sLineBreak +
'用户输入：' + AInput;
end;

end.
```

### 4.4 领域模型（NL解析结果类）

```delphi
unit Domain.NL;

interface

uses
  System.SysUtils, System.JSON, System.Generics.Collections,
  Domain.Entities;  // 复用现有 TUIScreen 类

type
  /// <summary>表单字段</summary>
  TNLField = record
    Name: string;       // 字段名（英文）
    FieldType: string;  // 字段类型: text/password/email/phone/textarea
    Caption: string;    // 显示标签
    Required: Boolean;  // 是否必填
  end;

  /// <summary>提交按钮</summary>
  TNLButton = record
    Text: string;   // 按钮文字
    Style: string;  // 样式: primary/secondary
  end;

  /// <summary>NL解析结果</summary>
  TNLParseResult = class
  private
    FResultType: string;
    FTitle: string;
    FFields: TArray<TNLField>;
    FSubmitButton: TNLButton;
  public
    class function FromJSON(const AJSON: string): TNLParseResult;
    function ToUIScreen: TUIScreen;  // 转换为 DevDirector 现有实体

    property ResultType: string read FResultType;
    property Title: string read FTitle;
    property Fields: TArray<TNLField> read FFields;
    property SubmitButton: TNLButton read FSubmitButton;
  end;

implementation

class function TNLParseResult.FromJSON(const AJSON: string): TNLParseResult;
var
  LRoot: TJSONObject;
  LFields: TJSONArray;
  LField: TJSONObject;
  LBtn: TJSONObject;
  I: Integer;
begin
  Result := TNLParseResult.Create;
  LRoot := TJSONObject.ParseJSONValue(AJSON) as TJSONObject;
  try
    if not Assigned(LRoot) then
      raise Exception.Create('Invalid JSON response');
      
    Result.FResultType := LRoot.GetValue<string>('type', 'FormPage');
    Result.FTitle := LRoot.GetValue<string>('title', '');

    LFields := LRoot.GetValue<TJSONArray>('fields');
    if Assigned(LFields) then
    begin
      SetLength(Result.FFields, LFields.Count);
      for I := 0 to LFields.Count - 1 do
      begin
        LField := LFields.Items[I] as TJSONObject;
        Result.FFields[I].Name := LField.GetValue<string>('name', '');
        Result.FFields[I].FieldType := LField.GetValue<string>('type', 'text');
        Result.FFields[I].Caption := LField.GetValue<string>('label', '');
        Result.FFields[I].Required := LField.GetValue<Boolean>('required', False);
      end;
    end;

    LBtn := LRoot.GetValue<TJSONObject>('submitButton');
    if Assigned(LBtn) then
    begin
      Result.FSubmitButton.Text := LBtn.GetValue<string>('text', '提交');
      Result.FSubmitButton.Style := LBtn.GetValue<string>('style', 'primary');
    end
    else
    begin
      Result.FSubmitButton.Text := '提交';
      Result.FSubmitButton.Style := 'primary';
    end;
  finally
    LRoot.Free;
  end;
end;

function TNLParseResult.ToUIScreen: TUIScreen;
begin
  // 转换为 DevDirector 的 TUIScreen 实体
  // 具体实现依赖 Domain.Entities 的定义
  Result := TUIScreen.Create;
  Result.Title := FTitle;
  // ... 映射字段到控件
end;

end.
```

### 6.2 重试机制

```delphi
procedure TNLParserCtrl.ParseWithRetry(
  const AInput: string;
  const ASceneType: string;
  const AMaxRetries: Integer;
  const OnSuccess: TProc<TNLResult>;
  const OnError: TProc<string>);
var
  LRetryCount: Integer;
  LLastError: string;
begin
  LRetryCount := 0;
  
  while LRetryCount <= AMaxRetries do
  begin
    ParseAsync(AInput, ASceneType,
      procedure(AResult: TNLResult)
      var
        LValidation: TNLValidationResult;
      begin
        LValidation := FValidator.Validate(AResult);
        
        if LValidation.Valid then
        begin
          if Assigned(OnSuccess) then OnSuccess(AResult);
        end
        else
        begin
          Inc(LRetryCount);
          if LRetryCount <= AMaxRetries then
          begin
            // 重试，带上错误信息
            LLastError := String.Join(', ', LValidation.Errors);
            Log.Warn('验证失败，第%d次重试 %s', [LRetryCount, LLastError]);
          end
          else
          begin
            if Assigned(OnError) then 
              OnError('无法生成有效输出: ' + LLastError);
          end;
        end;
      end,
      procedure(AError: string)
      begin
        LLastError := AError;
        Inc(LRetryCount);
        if LRetryCount > AMaxRetries then
        begin
          if Assigned(OnError) then OnError(AError);
        end;
      end);
  end;
end;
```

|----------|------|------|
| 提示词 | `prompts` | AI提示词模板，带版本号 |
| 翻译文本 | `translations` | i18n多语言文本 |
| 配置参数 | `configs` | 运行时配置 |
| 规则定义 | `rules` | 业务规则、验证规则 |

**prompts 表设计？**

```sql
CREATE TABLE IF NOT EXISTS prompts (
  id TEXT PRIMARY KEY,
  category TEXT NOT NULL,        -- 分类: nl_system, nl_user, etc.
  scene_type TEXT NOT NULL,      -- 场景: FormPage, ListPage, etc.
  version INTEGER NOT NULL,      -- 版本号
  content TEXT NOT NULL,         -- 提示词内容
  is_active BOOLEAN DEFAULT 0,   -- 是否启用
  created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
  remark TEXT,                   -- 备注
  
  UNIQUE(category, scene_type, version)
);

CREATE INDEX idx_prompts_active ON prompts(category, scene_type, is_active);
```

**提示词加载示例？**

```delphi
// ✅ 正确：从数据库加载，修改提示词无需重新编译
function TPromptRepository.GetActivePrompt(
  const ACategory, ASceneType: string): TPromptRecord;
begin
  Result := FDb.QueryOne<TPromptRecord>(
    'SELECT * FROM prompts WHERE category = ? AND scene_type = ? AND is_active = 1',
    [ACategory, ASceneType]);
end;

// ❌ 错误：硬编码提示词，每次修改都要重新编译
const
  SYSTEM_PROMPT = '你是一个...';
```

**优势说明：**

1. **快速迭代**：修改提示词后立即生效，无需重新编译和部署
2. **版本管理**：可回滚到任意历史版本
3. **A/B测试**：可同时保留多个版本，根据条件切换
4. **审计追溯**：记录每次变更
5. **非技术人员可编辑**：业务人员可直接修改提示词

### 9.3 设计反模式避免

| 反模式| 说明 | 危害 |
|----------|------|------|
| **硬编码提示词** | 提示词写死在代码中| 修改需重新编译、无法版本控制|
| 硬编码翻译| i18n文本写在代码中| 多语言维护困难 |
| 魔法数字 | 配置值写死在代码中| 调整需重新编译 |
| 无限重试 | AI调用无重试上限| 可能死循环、耗尽预算 |
| 同步阻塞 | AI调用阻塞主线程| UI卡死 |
| 无验证直接使用| AI输出不经验证直接使用 | 幻觉传播到业务层 |

## 附录：开发文档编写规范

### A.1 核心原则：聚焦产品内核？

开发文档的核心价值是让读者理解*为什么这么做**？

### A.2 四条必须遵循的规则？

1. **先说为什么，再说怎么做？**
   - 每个模块/功能先解释存在的理由
   - 设计决策的背景和权衡

2. **强调产品内核，而非时间线？**
   - 避免 "Week 1 做X，Week 2 做Y" 的表格？
   - 使用 "先完成X，X 稳定后做 Y" 的表格？

3. **包含正确/错误示例对比**
   - 每个重要设计点都要有示例
   - 务必明确标记正确和错误的做法

4. **可结典化**
   - 文档应能成为团队的知识资源？
   - 新人能通过文档理解为什么这样设计？

### A.3 文档结构模板

```markdown
## X、模块名称？

### X.1 为什么需要这个模块？
解决什么问题？带来什么价值？

### X.2 设计决策
为什么选择这个方案？考虑过哪些替代方案？

### X.3 实现要点
代码示例 + 正确/错误对比

### X.4 依赖与边界？
与其他模块的关系
```

### A.4 示例对比

**例如错误（纯时间线描述）**？

```
Week 1: 完成NL解析模块？
Week 2: 完成验证模块？
Week 3: 完成UI
```

**错误**正确（强调内核）**错误**

```
NL解析器是核心，确保80% 首次正确率后再进入下一步即可
验证器是安全网，必须与解析器同步开发才可
UI只是展示层，在核心稳定后进行即可
```
