```typescript
interface CacheHit {
  type: 'exact' | 'semantic' | 'miss';
  result?: StructuredOutput;
  confidence?: number;
  source?: string;
}

async function queryCache(input: string, context: Context): Promise<CacheHit> {
  // 1. 精确匹配（最快）
  const inputHash = hash(normalize(input) + context.domain);
  const exactHit = await db.query(
    'SELECT * FROM conversation_cache WHERE input_hash = ? AND is_valid = 1',
    [inputHash]
  );
  if (exactHit) {
    await db.run('UPDATE conversation_cache SET hit_count = hit_count + 1, last_hit_at = ? WHERE id = ?', 
                 [now(), exactHit.id]);
    return { type: 'exact', result: JSON.parse(exactHit.output_json), confidence: 1.0 };
  }
  
  // 2. 语义相似搜索
  const embedding = await getEmbedding(input);
  const similarHits = await vectorSearch(embedding, {
    threshold: 0.92,
    limit: 3,
    filter: { context_domain: context.domain }
  });
  
  if (similarHits.length > 0) {
    const best = similarHits[0];
    // 相似度很高且上下文匹配，直接使用
    if (best.similarity > 0.95 && best.context_hash === context.hash) {
      return { type: 'semantic', result: best.output, confidence: best.similarity };
    }
    // 相似度一般，作为参�?
    if (best.similarity > 0.85) {
      return { type: 'semantic', result: best.output, confidence: best.similarity, needsVerify: true };
    }
  }
  
  // 3. 未命�?
  return { type: 'miss' };
}
```

```typescript
// 合并多个小请求为一个大请求

class AIBatcher {
  private queue: AIRequest[] = [];
  private timer: NodeJS.Timeout | null = null;
  
  async request(req: AIRequest): Promise<AIResponse> {
    return new Promise((resolve) => {
      this.queue.push({ ...req, resolve });
      
      if (!this.timer) {
        this.timer = setTimeout(() => this.flush(), 50); // 50ms内的请求合并
      }
    });
  }
  
  private async flush(): Promise<void> {
    const batch = this.queue.splice(0);
    this.timer = null;
    
    if (batch.length === 1) {
      // 单个请求直接�?
      const result = await ai.call(batch[0]);
      batch[0].resolve(result);
    } else {
      // 多个请求合并
      const combinedPrompt = batch.map((r, i) => `[${i}] ${r.prompt}`).join('\n---\n');
      const results = await ai.call({ prompt: combinedPrompt, expectMultiple: true });
      
      batch.forEach((req, i) => req.resolve(results[i]));
    }
  }
}
```
