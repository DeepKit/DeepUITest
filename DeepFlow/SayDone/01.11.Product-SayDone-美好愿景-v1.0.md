  LAgent: IAgent;
begin
  LBlackboard := TBlackboard.Create(ATask);
  try
    LAgents := SelectAgents(ATask);
    
    // 并行工作
    SetLength(LTasks, Length(LAgents));
    for var I := 0 to High(LAgents) do

  end;
