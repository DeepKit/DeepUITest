```pascal
function TIPCClient.ExecuteWithRetry<T>(ARequest: TIPCRequest): T;

  Attempt: Integer;
  LastError: Exception;

  Attempt := 0;
  repeat
    Inc(Attempt);
    try

      Exit;  // 成功则返�?
```

```pascal
type
  TCircuitState = (csClose, csOpen, csHalfOpen);

  TCircuitBreaker = class
  private
    FState: TCircuitState;
    FFailureCount: Integer;
    FLastFailureTime: TDateTime;
    FThreshold: Integer;
    FTimeout: Integer;
    FLock: TCriticalSection;
    
    procedure TransitionTo(ANewState: TCircuitState);
    function ShouldAttemptReset: Boolean;
  public
    constructor Create(AThreshold: Integer = 5; ATimeout: Integer = 30000);
    destructor Destroy; override;
    
    // 检查是否允许请求通过
    function AllowRequest: Boolean;
    
    // 记录成功
    procedure RecordSuccess;
    
    // 记录失败
    procedure RecordFailure;
    
    // 当前状�?
    property State: TCircuitState read FState;
    property FailureCount: Integer read FFailureCount;
  end;

function TCircuitBreaker.AllowRequest: Boolean;
begin
  FLock.Enter;
  try
    case FState of
      csClose:
        Result := True;
      
      csOpen:
        if ShouldAttemptReset then
        begin
          TransitionTo(csHalfOpen);
          Result := True;  // 允许试探请求
        end
        else
          Result := False;
      
      csHalfOpen:
        Result := True;  // 半开状态允许请�?
    end;
  finally
    FLock.Leave;
  end;
end;

procedure TCircuitBreaker.RecordSuccess;
begin
  FLock.Enter;
  try
    FFailureCount := 0;
    if FState = csHalfOpen then
      TransitionTo(csClose);
  finally
    FLock.Leave;
  end;
end;

procedure TCircuitBreaker.RecordFailure;
begin
  FLock.Enter;
  try
    Inc(FFailureCount);
    FLastFailureTime := Now;
    
    case FState of
      csClose:
        if FFailureCount >= FThreshold then
          TransitionTo(csOpen);
      
      csHalfOpen:
        TransitionTo(csOpen);  // 半开状态失败立即熔�?
    end;
  finally
    FLock.Leave;
  end;
end;
```

```python
class ExceptionContext:
    """异常上下�?""
    def __init__(self):
        self.request_id: Optional[str] = None
        self.trace_id: Optional[str] = None
        self.skill_name: Optional[str] = None
        self.user_id: Optional[str] = None

    """设置当前请求的异常上下文"""
    ctx = get_current_context()
    for key, value in kwargs.items():
        if hasattr(ctx, key):
            setattr(ctx, key, value)
        else:

class GlobalExceptionHandler:
    """全局异常处理�?""
    
    def __init__(self, on_fatal: Callable = None):
        self.on_fatal = on_fatal
        self._install_handlers()
    
    def _install_handlers(self):
        """安装全局异常处理"""
        # 未捕获异�?
        sys.excepthook = self._handle_uncaught_exception
        
        # 信号处理
        signal.signal(signal.SIGTERM, self._handle_sigterm)
        signal.signal(signal.SIGINT, self._handle_sigint)
        
        # 线程异常（Python 3.8+�?
        if hasattr(threading, 'excepthook'):
            threading.excepthook = self._handle_thread_exception
    
    def _handle_uncaught_exception(self, exc_type, exc_value, exc_tb):
        """处理未捕获的异常"""
        if issubclass(exc_type, KeyboardInterrupt):
            sys.__excepthook__(exc_type, exc_value, exc_tb)
            return
        
        report = ExceptionReport(exc_value, ExceptionCategory.PROCESS_CRASH)
        logger.critical(f"Uncaught exception: {report.to_json()}")
        
        # 写入崩溃日志文件
        self._write_crash_log(report)
        
        # 通知主进�?
        if self.on_fatal:
            self.on_fatal(report)
    
    def _handle_thread_exception(self, args):
        """处理线程中的未捕获异�?""
        report = ExceptionReport(args.exc_value, ExceptionCategory.REQUEST_FAILURE)
        logger.error(f"Thread exception in {args.thread.name}: {report.to_json()}")
    
    def _handle_sigterm(self, signum, frame):
        """处理 SIGTERM 信号"""
        logger.info("Received SIGTERM, initiating graceful shutdown...")
        self._graceful_shutdown()
    
    def _handle_sigint(self, signum, frame):
        """处理 SIGINT 信号 (Ctrl+C)"""
        logger.info("Received SIGINT, initiating graceful shutdown...")
        self._graceful_shutdown()
    
    def _graceful_shutdown(self):
        """优雅关闭"""
        # 1. 停止接受新请�?
        # 2. 等待进行中请求完�?
        # 3. 保存状态（如果需要）
        # 4. 释放资源
        # 5. 退�?
        logger.info("Graceful shutdown complete")
        sys.exit(0)
    
    def _write_crash_log(self, report: ExceptionReport):
        """写入崩溃日志"""
        try:
            crash_log_path = f"crash_{int(report.timestamp)}.log"
            with open(crash_log_path, 'w', encoding='utf-8') as f:
                f.write(report.to_json())
            logger.info(f"Crash log written to {crash_log_path}")
        except Exception as e:
            logger.error(f"Failed to write crash log: {e}")


def with_exception_handling(category: ExceptionCategory = ExceptionCategory.SKILL_ERROR):
    """Skill 级异常处理装饰器"""
    def decorator(func: Callable) -> Callable:
        @wraps(func)
        def wrapper(*args, **kwargs):
            try:
                return func(*args, **kwargs)
            except Exception as e:
                report = ExceptionReport(e, category)
                logger.error(f"Exception in {func.__name__}: {report.to_json()}")
                
                # 返回标准错误响应
                return {

                    "error": {
                        "code": _get_error_code(category),
                        "message": str(e),
                        "category": category.value,
                        "request_id": get_current_context().request_id
                    }
                }
        return wrapper
    return decorator
```
