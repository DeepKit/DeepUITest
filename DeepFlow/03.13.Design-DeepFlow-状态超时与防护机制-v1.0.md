### 3.2 检测查�?

```sql
-- 查找状态超时的流程
SELECT 
  f.flow_id,
  f.status,
  f.flow_type,
  e.timestamp as last_event_time,
  e.step as last_step
FROM flow_instances f
JOIN (
  SELECT flow_id, MAX(timestamp) as timestamp, step
  FROM uni_flow_events
  GROUP BY flow_id
) e ON f.flow_id = e.flow_id
WHERE f.status NOT IN ('Succeeded', 'Failed', 'Cancelled')
  AND (
    -- Created 超时
    (f.status = 'Created' AND e.timestamp < NOW() - INTERVAL '5 minutes')
    OR
    -- Running 超时
    (f.status = 'Running' AND e.timestamp < NOW() - INTERVAL '30 minutes')
    OR
    -- WaitingUser 超时
    (f.status = 'WaitingUser' AND e.timestamp < NOW() - INTERVAL '24 hours')
  )
ORDER BY e.timestamp ASC
LIMIT 100;
```

## 4. 超时处理策略

### 4.1 分级处理

```ts
interface TimeoutHandler {
  // 预警（达到阈值）
  onWarning(flow: FlowInstance): Promise<void>;
  
  // 软超时（首次超时�?
  onSoftTimeout(flow: FlowInstance): Promise<void>;
  
  // 硬超时（超时后仍无响应）
  onHardTimeout(flow: FlowInstance): Promise<void>;
}

class DefaultTimeoutHandler implements TimeoutHandler {
  async onWarning(flow: FlowInstance): Promise<void> {
    // 1. 记录警告事件
    await this.writeEvent(flow.flowId, {
      type: flow.type,
      step: '__system__',
      source: 'uniFlow/TimeoutGuard',
      status: 'Started',
      payload: {
        warningType: 'TIMEOUT_APPROACHING',
        currentStatus: flow.status,
        timeRemaining: this.getTimeRemaining(flow),
      },
    });
    
    // 2. 发送通知（如果配置了�?
    await this.notifyUser(flow, 'WARNING');
  }
  
  async onSoftTimeout(flow: FlowInstance): Promise<void> {
    // 根据状态决定处理方�?
    switch (flow.status) {
      case 'Created':
        // 未启动的流程直接取消
        await this.cancelFlow(flow, 'CREATED_TIMEOUT');
        break;
        
      case 'Running':
        // 运行中的流程先暂�?
        await this.suspendFlow(flow, 'RUNNING_TIMEOUT');
        break;
        
      case 'WaitingUser':
        // 等待用户的流程发送提�?
        await this.notifyUser(flow, 'SOFT_TIMEOUT');
        break;
    }
  }
  
  async onHardTimeout(flow: FlowInstance): Promise<void> {
    // 强制取消流程
    await this.forceCancel(flow, 'HARD_TIMEOUT');
  }
}
```

### 5.2 心跳实现

```ts
class HeartbeatManager {
  private heartbeats: Map<UniFlowId, Heartbeat> = new Map();
  
  // 发送心�?
  async sendHeartbeat(
    flowId: UniFlowId, 
    step: string, 
    progress?: number
  ): Promise<void> {
    const heartbeat: Heartbeat = {
      flowId,
      timestamp: new Date(),
      step,
      progress,
    };
    
    this.heartbeats.set(flowId, heartbeat);
    
    // 可选：持久化心跳（用于跨进程恢复）
    await this.persistHeartbeat(heartbeat);
  }
  
  // 检查心跳是否存�?
  isAlive(flowId: UniFlowId, config: HeartbeatConfig): boolean {
    const heartbeat = this.heartbeats.get(flowId);
    if (!heartbeat) return false;
    
    const elapsed = Date.now() - heartbeat.timestamp.getTime();
    const maxMissed = config.interval * config.missedThreshold;
    
    return elapsed < maxMissed;
  }
}
```

### 5.3 带心跳的长时间操�?

```ts
async function executeWithHeartbeat<T>(
  flowId: UniFlowId,
  step: string,
  operation: () => Promise<T>,
  heartbeatInterval: number = 30000
): Promise<T> {
  const heartbeatManager = getHeartbeatManager();
  
  // 启动心跳
  const heartbeatTimer = setInterval(() => {
    heartbeatManager.sendHeartbeat(flowId, step);
  }, heartbeatInterval);
  
  try {
    // 执行实际操作
    return await operation();
  } finally {
    // 停止心跳
    clearInterval(heartbeatTimer);
  }
}
```

## 6. 系统重启恢复

### 6.1 启动时扫�?

```ts
interface StartupRecovery {
  // 系统启动时调�?
  async recover(): Promise<RecoveryReport>;
}

interface RecoveryReport {
  scannedFlows: number;
  resumedFlows: number;
  cancelledFlows: number;
  errors: RecoveryError[];
}

class DefaultStartupRecovery implements StartupRecovery {
  async recover(): Promise<RecoveryReport> {
    const report: RecoveryReport = {
      scannedFlows: 0,
      resumedFlows: 0,
      cancelledFlows: 0,
      errors: [],
    };
    
    // 1. 查找所有未完成的流�?
    const pendingFlows = await this.findPendingFlows();
    report.scannedFlows = pendingFlows.length;
    
    for (const flow of pendingFlows) {
      try {
        // 2. 根据状态决定恢复策�?
        if (this.canResume(flow)) {
          await this.resumeFlow(flow);
          report.resumedFlows++;
        } else {
          await this.cancelFlow(flow, 'SYSTEM_RESTART_RECOVERY');
          report.cancelledFlows++;
        }
      } catch (error) {
        report.errors.push({
          flowId: flow.flowId,
          error: error.message,
        });
      }
    }
    
    return report;
  }
  
  private canResume(flow: FlowInstance): boolean {
    // WaitingUser 状态可以安全恢�?
    if (flow.status === 'WaitingUser') return true;
    
    // 其他状态根据最后事件判�?
    const lastEvent = this.getLastEvent(flow.flowId);
    const timeSinceLastEvent = Date.now() - new Date(lastEvent.timestamp).getTime();
    
    // 如果最后事件在 5 分钟内，尝试恢复
    return timeSinceLastEvent < 5 * 60 * 1000;
  }
}
```
