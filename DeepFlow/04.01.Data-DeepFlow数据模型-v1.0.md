CREATE TABLE skills (

    name TEXT NOT NULL UNIQUE,        -- [L1] Skill 名称
    version TEXT NOT NULL,            -- [L1] 版本�?
    description TEXT,                 -- [L1] 描述

    id TEXT PRIMARY KEY,              -- [L1] UUID

    node_id TEXT,                     -- [L1] 关联的节�?

    error_message TEXT,               -- [L2] 可能含敏感上下文

    finished_at TIMESTAMP,            -- [L0]
