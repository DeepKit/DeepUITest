class APIKeyManager:
    def create_key(self, name: str, permissions: List[str], 
                   expires_at: Optional[datetime] = None) -> APIKey:
        """创建新的 API Key"""

    def rotate_key(self, key_id: str, grace_period: int = 86400) -> APIKey:
        """轮换 API Key（旧 Key 在宽限期内仍可用�?""
        old_key = self.store.get(key_id)
        old_key.deprecated_at = datetime.utcnow()
        old_key.expires_at = datetime.utcnow() + timedelta(seconds=grace_period)
        
        new_key = self.create_key(
            name=old_key.name,
            permissions=old_key.permissions
        )
        return new_key
    
    def revoke_key(self, key_id: str) -> None:
        """立即吊销 API Key"""
        key = self.store.get(key_id)
        key.revoked = True
        key.revoked_at = datetime.utcnow()
        self.store.save(key)

## 3. 请求签名

### 3.1 签名算法

对于 L2+ 安全等级，所有写操作必须携带签名�?

**签名参数**�?

| 参数 | Header | 说明 |
|------|--------|------|
| API Key | `X-API-Key` | 身份标识 |
| 时间�?| `X-Timestamp` | Unix 时间戳（秒） |
| 随机�?| `X-Nonce` | 一次性随机字符串 |
| 签名 | `X-Signature` | 计算得到的签名�?|

**签名计算**�?

```python
import hmac
import hashlib
import time
import uuid

def generate_signature(
    api_key: str,
    api_secret: str,
    method: str,
    path: str,
    timestamp: int,
    nonce: str,
    body: str = ""
) -> str:
    """
    生成请求签名
    
    签名字符串格式：
    {method}\n{path}\n{timestamp}\n{nonce}\n{body_hash}
    """
    # 计算请求体的 SHA256 哈希
    body_hash = hashlib.sha256(body.encode()).hexdigest() if body else ""
    
    # 构建签名字符�?
    string_to_sign = f"{method.upper()}\n{path}\n{timestamp}\n{nonce}\n{body_hash}"
    
    # HMAC-SHA256 签名
    signature = hmac.new(
        api_secret.encode(),
        string_to_sign.encode(),
        hashlib.sha256
    ).hexdigest()
    
    return signature

# 使用示例
def make_signed_request(
    method: str,
    url: str,
    body: dict = None,
    api_key: str = "",
    api_secret: str = ""
):
    from urllib.parse import urlparse
    
    parsed = urlparse(url)
    path = parsed.path
    timestamp = int(time.time())
    nonce = str(uuid.uuid4())
    body_str = json.dumps(body) if body else ""
    
    signature = generate_signature(
        api_key=api_key,
        api_secret=api_secret,
        method=method,
        path=path,
        timestamp=timestamp,
        nonce=nonce,
        body=body_str
    )
    
    headers = {
        "X-API-Key": api_key,
        "X-Timestamp": str(timestamp),
        "X-Nonce": nonce,
        "X-Signature": signature,
        "Content-Type": "application/json"
    }
    
    return requests.request(method, url, headers=headers, json=body)
```

### 3.2 服务端签名验�?

```python
from datetime import datetime, timedelta

class SignatureValidator:
    # 时间戳有效窗口（秒）
    TIMESTAMP_TOLERANCE = 300  # 5 分钟
    
    # Nonce 缓存过期时间（秒�?
    NONCE_TTL = 600  # 10 分钟
    
    def __init__(self, nonce_store: NonceStore, key_store: KeyStore):
        self.nonce_store = nonce_store
        self.key_store = key_store
    
    def validate(self, request) -> Tuple[bool, str]:
        """
        验证请求签名
        返回: (是否有效, 错误信息)
        """
        # 1. 提取头部
        api_key = request.headers.get("X-API-Key")
        timestamp = request.headers.get("X-Timestamp")
        nonce = request.headers.get("X-Nonce")
        signature = request.headers.get("X-Signature")
        
        if not all([api_key, timestamp, nonce, signature]):
            return False, "Missing required headers"
        
        # 2. 验证时间戳（防止重放�?
        try:
            ts = int(timestamp)
            now = int(time.time())
            if abs(now - ts) > self.TIMESTAMP_TOLERANCE:
                return False, "Timestamp expired"
        except ValueError:
            return False, "Invalid timestamp"
        
        # 3. 验证 Nonce 唯一性（防止重放�?
        if self.nonce_store.exists(nonce):
            return False, "Nonce already used"
        
        # 4. 获取 API Secret
        key_info = self.key_store.get(api_key)
        if not key_info:
            return False, "Invalid API key"
        
        # 5. 计算预期签名
        expected = generate_signature(
            api_key=api_key,
            api_secret=key_info.secret,
            method=request.method,
            path=request.path,
            timestamp=ts,
            nonce=nonce,
            body=request.data.decode() if request.data else ""
        )
        
        # 6. 使用时间安全比较
        if not hmac.compare_digest(expected, signature):
            return False, "Invalid signature"
        
        # 7. 记录 Nonce（防止重复使用）
        self.nonce_store.add(nonce, ttl=self.NONCE_TTL)
        
        return True, ""
```

from typing import Optional, Dict, Any
import re

import re

def mask_sensitive(text: str) -> str:
    """脱敏敏感信息"""
    patterns = [
        (r'(api[_-]?key["\s:=]+)["\']?([^"\'\s]+)', r'\1***'),
        (r'(password["\s:=]+)["\']?([^"\'\s]+)', r'\1***'),

        result = re.sub(pattern, replacement, result, flags=re.IGNORECASE)

SENSITIVE_FIELDS = {'password', 'secret', 'token', 'api_key', 'credit_card'}

def redact_response(data: dict) -> dict:
    """移除响应中的敏感字段"""

        if isinstance(obj, dict):
            return {
                k: '***' if k.lower() in SENSITIVE_FIELDS else redact(v)
                for k, v in obj.items()
            }
        elif isinstance(obj, list):
