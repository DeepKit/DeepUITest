```python
# evaluation/deterministic.py

def exact_match(output: str, expected: str) -> bool:
    """精确匹配"""
    return output.strip().lower() == expected.strip().lower()


def contains_all(output: str, keywords: list) -> bool:
    """必须包含所有关键词"""
    output_lower = output.lower()
    return all(kw.lower() in output_lower for kw in keywords)


def contains_none(output: str, forbidden: list) -> bool:
    """不能包含任何禁止�?""
    output_lower = output.lower()
    return all(kw.lower() not in output_lower for kw in forbidden)


def json_structure_match(output: str, schema: dict) -> bool:
    """JSON 结构匹配"""
    import json
    from jsonschema import validate, ValidationError
    
    try:
        data = json.loads(output)
        validate(instance=data, schema=schema)
        return True
    except (json.JSONDecodeError, ValidationError):
        return False


def length_check(output: str, min_len: int = 0, max_len: int = None) -> bool:
    """长度检�?""
    length = len(output)
    if length < min_len:
        return False
    if max_len and length > max_len:
        return False
    return True
```

```python
# evaluation/semantic.py

from sentence_transformers import SentenceTransformer
from sklearn.metrics.pairwise import cosine_similarity
import numpy as np

class SemanticEvaluator:
    """语义相似度评估器"""
    
    def __init__(self, model_name: str = "paraphrase-multilingual-MiniLM-L12-v2"):
        self.model = SentenceTransformer(model_name)
    
    def similarity(self, text1: str, text2: str) -> float:
        """计算两个文本的语义相似度"""
        embeddings = self.model.encode([text1, text2])
        similarity = cosine_similarity([embeddings[0]], [embeddings[1]])[0][0]
        return float(similarity)
    
    def evaluate(self, output: str, reference: str, threshold: float = 0.7) -> dict:
        """评估输出与参考答案的相似�?""
        score = self.similarity(output, reference)
        return {
            "score": score,
            "passed": score >= threshold,
            "threshold": threshold
        }
    
    def batch_evaluate(self, outputs: list, references: list, threshold: float = 0.7) -> list:
        """批量评估"""
        results = []
        for output, reference in zip(outputs, references):
            results.append(self.evaluate(output, reference, threshold))
        return results


# 使用示例
evaluator = SemanticEvaluator()
result = evaluator.evaluate(
    output="今天北京天气不错，气温大�?度左�?,
    reference="北京今天天气晴朗，温度约5-10�?,
    threshold=0.7
)
# result: {"score": 0.85, "passed": True, "threshold": 0.7}
```

### 4.4 综合评估框架

```python
# evaluation/composite.py

from dataclasses import dataclass
from typing import List, Dict, Any, Optional
from enum import Enum

class EvaluationType(Enum):
    EXACT_MATCH = "exact_match"
    CONTAINS = "contains"
    SEMANTIC = "semantic"
    LLM_JUDGE = "llm_judge"
    JSON_SCHEMA = "json_schema"


@dataclass
class EvaluationResult:
    """评估结果"""
    test_id: str
    passed: bool
    score: float
    details: Dict[str, Any]
    evaluation_type: EvaluationType
    duration_ms: int


class CompositeEvaluator:
    """综合评估�?""
    
    def __init__(self):
        self.semantic_evaluator = SemanticEvaluator()
        self.llm_judge = None  # 可�?
    
    def evaluate(self, test_case: Dict, output: str) -> EvaluationResult:
        """根据测试用例配置进行评估"""
        expected = test_case["expected_output"]
        evaluation_config = test_case.get("evaluation", {})
        method = evaluation_config.get("method", "contains")
        
        start_time = time.time()
        
        if method == "exact_match":
            passed = exact_match(output, evaluation_config["expected"])
            score = 1.0 if passed else 0.0
            details = {"matched": passed}
            
        elif method == "contains":
            must_contain = expected.get("must_contain", [])
            must_not_contain = expected.get("must_not_contain", [])
            
            contains_pass = contains_all(output, must_contain)
            excludes_pass = contains_none(output, must_not_contain)
            
            passed = contains_pass and excludes_pass
            score = (0.5 if contains_pass else 0) + (0.5 if excludes_pass else 0)
            details = {
                "contains_all": contains_pass,
                "excludes_all": excludes_pass,
                "missing": [k for k in must_contain if k.lower() not in output.lower()],
                "forbidden_found": [k for k in must_not_contain if k.lower() in output.lower()]
            }
            
        elif method == "semantic_similarity":
            reference = evaluation_config["reference"]
            threshold = evaluation_config.get("threshold", 0.7)
            
            result = self.semantic_evaluator.evaluate(output, reference, threshold)
            passed = result["passed"]
            score = result["score"]
            details = result
            
        elif method == "llm_judge":
            if not self.llm_judge:
                raise ValueError("LLM Judge not configured")
            
            question = test_case["input"]["prompt"]
            reference = evaluation_config.get("reference", "")
            threshold = evaluation_config.get("threshold", 7.0)
            
            result = self.llm_judge.evaluate(question, output, reference, threshold)
            passed = result["passed"]
            score = result.get("overall_score", 0) / 10.0
            details = result
            
        elif method == "json_schema":
            schema = evaluation_config["schema"]
            passed = json_structure_match(output, schema)
            score = 1.0 if passed else 0.0
            details = {"schema_valid": passed}
            
        else:
            raise ValueError(f"Unknown evaluation method: {method}")
        
        # 额外检�?
        if expected.get("max_length"):
            if len(output) > expected["max_length"]:
                passed = False
                details["length_exceeded"] = True
        
        duration_ms = int((time.time() - start_time) * 1000)
        
        return EvaluationResult(
            test_id=test_case["id"],
            passed=passed,
            score=score,
            details=details,
            evaluation_type=EvaluationType(method),
            duration_ms=duration_ms
        )
```

```python
# runner.py

import json
import time
from pathlib import Path
from dataclasses import dataclass
from typing import List, Dict, Any

@dataclass
class TestRun:
    """测试运行结果"""
    run_id: str
    started_at: str
    finished_at: str
    total_tests: int
    passed_tests: int
    failed_tests: int
    skipped_tests: int
    pass_rate: float
    results: List[EvaluationResult]
    summary: Dict[str, Any]


class GoldenDatasetRunner:
    """Golden Dataset 测试运行�?""
    
    def __init__(self, ai_client, evaluator: CompositeEvaluator):
        self.ai_client = ai_client
        self.evaluator = evaluator
    
    def run(self, dataset_path: str, filter_tags: List[str] = None) -> TestRun:
        """运行测试"""
        # 加载数据�?
        with open(dataset_path, 'r', encoding='utf-8') as f:
            dataset = json.load(f)
        
        scenarios = dataset["scenarios"]
        
        # 过滤
        if filter_tags:
            scenarios = [s for s in scenarios if any(t in s.get("tags", []) for t in filter_tags)]
        
        run_id = f"run_{int(time.time())}"
        started_at = datetime.now().isoformat()
        results = []
        
        for scenario in scenarios:
            try:
                result = self._run_single(scenario)
                results.append(result)
            except Exception as e:
                results.append(EvaluationResult(
                    test_id=scenario["id"],
                    passed=False,
                    score=0.0,
                    details={"error": str(e)},
                    evaluation_type=EvaluationType.EXACT_MATCH,
                    duration_ms=0
                ))
        
        finished_at = datetime.now().isoformat()
        
        passed = sum(1 for r in results if r.passed)
        failed = sum(1 for r in results if not r.passed)
        
        return TestRun(
            run_id=run_id,
            started_at=started_at,
            finished_at=finished_at,
            total_tests=len(results),
            passed_tests=passed,
            failed_tests=failed,
            skipped_tests=0,
            pass_rate=passed / len(results) if results else 0,
            results=results,
            summary=self._generate_summary(results)
        )
    
    def _run_single(self, scenario: Dict) -> EvaluationResult:
        """运行单个测试"""
        input_data = scenario["input"]
        prompt = input_data["prompt"]
        context = input_data.get("context", {})
        
        # 调用 AI
        start_time = time.time()
        output = self.ai_client.query(prompt, context)
        ai_duration = int((time.time() - start_time) * 1000)
        
        # 评估
        result = self.evaluator.evaluate(scenario, output)
        result.details["ai_output"] = output
        result.details["ai_duration_ms"] = ai_duration
        
        return result
    
    def _generate_summary(self, results: List[EvaluationResult]) -> Dict:
        """生成摘要"""
        by_category = {}
        for r in results:
            category = r.test_id.split("_")[0]
            if category not in by_category:
                by_category[category] = {"passed": 0, "failed": 0}
            if r.passed:
                by_category[category]["passed"] += 1
            else:
                by_category[category]["failed"] += 1
        
        avg_score = sum(r.score for r in results) / len(results) if results else 0
        
        return {
            "by_category": by_category,
            "average_score": avg_score,
            "failed_tests": [r.test_id for r in results if not r.passed]
        }
```

```python
def test_consistency(ai_client, prompt: str, num_runs: int = 5, threshold: float = 0.8):
    """一致性测�?""
    outputs = []
    for _ in range(num_runs):
        output = ai_client.query(prompt)
        outputs.append(output)
    
    # 计算两两相似�?

    for i in range(len(outputs)):
        for j in range(i + 1, len(outputs)):

    avg_similarity = sum(similarities) / len(similarities)
    
    return {
        "passed": avg_similarity >= threshold,
        "average_similarity": avg_similarity,
        "min_similarity": min(similarities),
        "max_similarity": max(similarities),
        "outputs": outputs
    }
```

```python
    error_rate: float


async def run_performance_test(
    ai_client,
    prompts: List[str],
    concurrency: int = 10,
    duration_seconds: int = 60
) -> PerformanceResult:
    """运行性能测试"""
    
    results = []
    errors = 0
    start_time = time.time()
    
    async def worker(prompt_queue: asyncio.Queue):
        nonlocal errors
        while time.time() - start_time < duration_seconds:
            try:
                prompt = await asyncio.wait_for(prompt_queue.get(), timeout=1.0)
            except asyncio.TimeoutError:
                continue
            
            req_start = time.time()
            try:
                await ai_client.query_async(prompt)
                latency = time.time() - req_start
                results.append(latency)
            except Exception:
                errors += 1
    
    # 创建任务队列
    queue = asyncio.Queue()
    
    # 填充队列
    async def producer():
        idx = 0
        while time.time() - start_time < duration_seconds:
            await queue.put(prompts[idx % len(prompts)])
            idx += 1
            await asyncio.sleep(0.01)
    
    # 启动工作协程
    workers = [asyncio.create_task(worker(queue)) for _ in range(concurrency)]
    producer_task = asyncio.create_task(producer())
    
    # 等待完成
    await asyncio.sleep(duration_seconds)
    producer_task.cancel()
    for w in workers:
        w.cancel()
    
    # 计算统计
    sorted_latencies = sorted(results)
    total = len(results) + errors
    
    return PerformanceResult(
        total_requests=total,
        successful_requests=len(results),
        failed_requests=errors,
        duration_seconds=duration_seconds,
        qps=len(results) / duration_seconds,
        latencies=results,
        p50=sorted_latencies[int(len(sorted_latencies) * 0.5)] if results else 0,
        p95=sorted_latencies[int(len(sorted_latencies) * 0.95)] if results else 0,
        p99=sorted_latencies[int(len(sorted_latencies) * 0.99)] if results else 0,
        error_rate=errors / total if total > 0 else 0
    )
```
