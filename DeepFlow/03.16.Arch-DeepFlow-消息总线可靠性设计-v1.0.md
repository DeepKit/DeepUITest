```pascal
procedure TMessageBus.HandleDeliveryFailure(AMessage: TUniFlowMessage; AError: string);
var
  ShouldRetry: Boolean;
  NextRetryDelay: Integer;
begin
  ShouldRetry := (AMessage.RetryCount < AMessage.MaxRetries) and
                 IsRetryableError(AError);
  
  if ShouldRetry then
  begin
    Inc(AMessage.FRetryCount);
    NextRetryDelay := CalculateBackoff(AMessage.RetryCount);
    
    Log.Warn('Message retry scheduled: msgId=%s, attempt=%d/%d, delay=%dms',
      [AMessage.MsgId, AMessage.RetryCount, AMessage.MaxRetries, NextRetryDelay]);
    
    // 延迟重新入队
    FRetryScheduler.Schedule(AMessage, NextRetryDelay);
  end
  else
  begin
    Log.Error('Message delivery failed permanently: msgId=%s, error=%s',
      [AMessage.MsgId, AError]);
    
    // 移入死信队列
    MoveToDeadLetter(AMessage, AError);
    
    // 触发告警
    FAlertManager.Alert(TAlertLevel.alError,
      Format('Message failed: %s', [AMessage.MsgId]));
  end;
end;

function TMessageBus.CalculateBackoff(AAttempt: Integer): Integer;
begin
  // 指数退避：1s, 2s, 4s, 8s... 最�?30s
  Result := Min(
    FConfig.BackoffInitial * Round(Power(FConfig.BackoffMultiplier, AAttempt - 1)),
    FConfig.BackoffMax
  );
  
  // 添加抖动（�?0%�?
  Result := Result + Random(Round(Result * 0.4)) - Round(Result * 0.2);
end;
```

type
  TRoleDeliveryState = (rdsHealthy, rdsDegraded, rdsFailed);

  TRoleHealthMonitor = class
  private
    FRoleStates: TDictionary<string, TRoleDeliveryState>;
    FFailureCounts: TDictionary<string, Integer>;
    FThreshold: Integer;
  public
    procedure RecordDeliveryResult(ARoleName: string; ASuccess: Boolean);

procedure TRoleHealthMonitor.RecordDeliveryResult(ARoleName: string; ASuccess: Boolean);
var
  FailCount: Integer;
begin
  if ASuccess then
  begin
    FFailureCounts[ARoleName] := 0;
    FRoleStates[ARoleName] := rdsHealthy;
  end
  else
  begin
    FailCount := FFailureCounts.GetValueOrDefault(ARoleName, 0) + 1;
    FFailureCounts[ARoleName] := FailCount;
    
    if FailCount >= FThreshold then
    begin
      FRoleStates[ARoleName] := rdsFailed;
      Log.Error('Role marked as failed: %s (failures: %d)', [ARoleName, FailCount]);
      
      // 触发角色故障事件
      PublishRoleFailureEvent(ARoleName);
    end
    else if FailCount >= FThreshold div 2 then
    begin
      FRoleStates[ARoleName] := rdsDegraded;
      Log.Warn('Role degraded: %s (failures: %d)', [ARoleName, FailCount]);
    end;
  end;
end;
