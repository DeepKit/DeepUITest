## 7. 安全防护

### 7.1 注入攻击防护

```python
class PromptSanitizer:
    """Prompt 注入防护"""
    
    # 危险模式
    DANGEROUS_PATTERNS = [
        r"ignore.*previous.*instructions",
        r"disregard.*above",
        r"forget.*everything",
        r"system.*prompt",
        r"<\|.*\|>",  # 特殊分隔�?
    ]
    
    @classmethod
    def sanitize(cls, user_input: str) -> str:
        """净化用户输�?""
        # 1. 检测危险模�?
        for pattern in cls.DANGEROUS_PATTERNS:
            if re.search(pattern, user_input, re.IGNORECASE):
                raise PromptInjectionDetected(f"检测到危险模式: {pattern}")
        
        # 2. 转义特殊字符
        user_input = cls.escape_special_chars(user_input)
        
        # 3. 长度限制
        if len(user_input) > MAX_USER_INPUT_LENGTH:
            user_input = user_input[:MAX_USER_INPUT_LENGTH] + "..."
        
        return user_input
    
    @staticmethod
    def escape_special_chars(text: str) -> str:
        """转义可能影响 Prompt 结构的字�?""
        replacements = {
            "{{": "{ {",
            "}}": "} }",
            "<|": "< |",
            "|>": "| >",
        }
        for old, new in replacements.items():
            text = text.replace(old, new)
        return text
```

class OutputValidator:
    """LLM 输出验证"""
    
    @classmethod
    def validate_json_output(cls, output: str, schema: dict) -> ValidationResult:
        """验证 JSON 输出格式"""
        try:
            # 提取 JSON 部分
            json_match = re.search(r'\{.*\}', output, re.DOTALL)
            if not json_match:
                return ValidationResult(False, "未找�?JSON 输出")
            
            data = json.loads(json_match.group())
            
            # Schema 验证
            jsonschema.validate(data, schema)
            
            return ValidationResult(True, data=data)
        except json.JSONDecodeError as e:
            return ValidationResult(False, f"JSON 解析错误: {e}")
        except jsonschema.ValidationError as e:
            return ValidationResult(False, f"Schema 验证失败: {e.message}")
