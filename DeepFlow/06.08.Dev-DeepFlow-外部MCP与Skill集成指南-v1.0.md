### 5.4 Skill 封装设计

#### 5.4.1 Python 端封装（示意�?

```python
from typing import Any, Dict
from uniflow_sdk.skill import UniFlowSkill
from mcp_client import MCPClient  # 假设�?MCP 客户端封�?

class DocSearchSkill(UniFlowSkill):
    @property
    def name(self) -> str:
        return "Skill.DocSearch"

    @property
    def version(self) -> str:
        return "1.0.0"

    async def execute(self, input: Dict[str, Any], context: Dict[str, Any]) -> Dict[str, Any]:
        query = input.get("query")
        ctx = input.get("context", {})

        limit = ctx.get("limit", 10)
        sources = ctx.get("sources", ["notion"])  # 默认 Notion

        client = MCPClient(server_id="docs-mcp-server")

        try:
            raw = await client.call_tool(
                tool="search_docs",
                arguments={
                    "query": query,
                    "limit": limit,
                    "source_filter": sources,
                    "include_snippet": True,
                },
                timeout=10,
            )
        except Exception as e:
            # 将外部错误映射为 UniFlow 统一错误格式
            return {
                "success": False,
                "error": f"DocSearch MCP error: {str(e)}",
                "items": []
            }

        items = []
        for r in raw.get("results", []):
            items.append({
                "title": r.get("title", ""),
                "snippet": r.get("snippet", ""),
                "url": r.get("url", ""),
                "source": r.get("source", ""),
                "score": r.get("score", 0.0),
                "metadata": {
                    "created_at": r.get("created_at"),
                    "updated_at": r.get("updated_at"),
                    "author": r.get("author"),
                },
            })

        return {
            "success": True,
            "items": items,
        }
```
